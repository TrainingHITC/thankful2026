    <script>
        // Link Web App của Google Apps Script
        const SCRIPT_URL = 'https://script.google.com/macros/s/AKfycbwz79YkGKZ2eLQJ-xyyR0TGKLqlvAFKEaT36Sbj_uZaU1SlWLC9wLFVuJwKv8dDYv4wuQ/exec';

        // Xử lý ẩn danh
        function toggleAnonymous() {
            const isAnon = document.getElementById('isAnonymous').checked;
            const senderInput = document.getElementById('senderName');
            if(isAnon) {
                senderInput.value = 'Một người giấu tên';
                senderInput.disabled = true;
                senderInput.classList.add('bg-gray-200');
            } else {
                senderInput.value = '';
                senderInput.disabled = false;
                senderInput.classList.remove('bg-gray-200');
            }
        }

        // Đóng mở Modal
        function openModal() { document.getElementById('writeModal').classList.add('modal-active'); }
        function closeModal() { document.getElementById('writeModal').classList.remove('modal-active'); }

        // Gửi dữ liệu về Google Sheets
        function submitLetter(e) {
            e.preventDefault();
            
            const btn = document.getElementById('submitBtn');
            const status = document.getElementById('statusText');

            btn.disabled = true;
            btn.innerHTML = 'Đang gửi thư... ⏳';
            
            const data = {
                receiver: document.getElementById('receiverName').value,
                sender: document.getElementById('senderName').value,
                message: document.getElementById('messageContent').value,
                isAnonymous: document.getElementById('isAnonymous').checked,
                envelopeTheme: document.querySelector('input[name="envelopeType"]:checked').value
            };

            fetch(SCRIPT_URL, {
                method: 'POST',
                body: JSON.stringify(data),
                headers: { 'Content-Type': 'text/plain;charset=utf-8' } // Tránh lỗi CORS preflight
            })
            .then(response => {
                status.classList.remove('hidden', 'text-red-500');
                status.classList.add('text-green-600');
                status.innerText = 'Thư của bạn đã được gửi thành công! 🎉';
                btn.innerHTML = 'Đã gửi ✓';
                
                setTimeout(() => {
                    closeModal();
                    document.getElementById('letterForm').reset();
                    btn.disabled = false;
                    btn.innerHTML = 'Gửi lá thư đi ✈️';
                    status.classList.add('hidden');
                }, 2000);
            })
            .catch(error => {
                status.classList.remove('hidden', 'text-green-600');
                status.classList.add('text-red-500');
                status.innerText = 'Có lỗi xảy ra, vui lòng thử lại sau.';
                btn.disabled = false;
                btn.innerHTML = 'Thử lại ✈️';
            });
        }
    </script>
